-- Commande de volet VELUX via KLF-50 et FGS-222
--Créer un VD dont le bouton principal fait monter ou descendre le volet en fonction d'un paramètre fonction de la dernière action avec timer terminé

local action =0
-- Action du bouton principal
-- If action = 1: lancer scénario Descente else lancer scénario Montée


-- Scénario Montée :
fibaro.call(S2,"turnOff")-- auto OFF prognammé à 50s
fibaro.call(S1,"turnOn") -- auto OFF prognammé à 50s
fibaro.set(action = 1)
-- Lancer le timer

-- Afficher l'Icône du mvt
 
-- Scénario Descente :
fibaro.call(S1,"turnOff")-- auto OFF prognammé à 50s
fibaro.call(S2,"turnOn") -- auto OFF prognammé à 50s

-- lancer le timer

-- Afficher l'Icône du mvt

-- Scénario Stop :
fibaro.call(S1,"turnOn")
fibaro.call(S2,"turnOn")
fibaro.set(action = 1)
-- Stopper le timer

-- Afficher Icone 2

fibaro.sleep(1000) -- attente de  1 avant de faire une RAZ des switchs S1 et S2 (Off)
fibaro.call(S1,"turnOff")-- Temporisation de RAZ
fibaro.call(S2,"turnOff")
